*////////////////////////////////////////////////// Triggers //////////////////////////////////////////////////////*
DELIMITER //
CREATE TRIGGER estado_auditar_insert
AFTER INSERT ON estado
FOR EACH ROW

begin
INSERT INTO auditoria_estado (accion, nuevo_id_estado, nuevo_nombre, usuario) 
VALUES ('Insertar', NEW.id_estado, NEW.nombre, current_user());
end //
DELIMITER ;

DELIMITER //
CREATE TRIGGER estado_auditar_delete
AFTER DELETE ON estado
FOR EACH ROW

begin
INSERT INTO auditoria_estado (accion, viejo_id_estado, viejo_nombre, usuario) 
VALUES ('Eliminar', old.id_estado, old.nombre, current_user());
end //
DELIMITER ;

DELIMITER //
CREATE TRIGGER estado_auditar_update
AFTER UPDATE ON estado
FOR EACH ROW

begin
INSERT INTO auditoria_estado (accion, nuevo_id_estado, viejo_id_estado, nuevo_nombre, viejo_nombre, usuario) 
VALUES ('Actualizar', new.id_estado, old.id_estado, new.nombre, old.nombre, current_user());
end //
DELIMITER ;

INSERT INTO estado VALUES(34, 'Prueba');
UPDATE estado SET id_estado = 34 ,nombre = 'P' WHERE estado.id_estado = 34;
DELETE FROM estado WHERE id_estado = 33;
SELECT * FROM auditoria_estado;
