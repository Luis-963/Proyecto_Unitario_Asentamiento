*////////////////////////////////////////////////////// BD //////////////////////////////////////////////////////////*
create database prueba_6to_1;
use prueba_6to_1;

create table estado(
 id_estado int(3) primary key,
 nombre varchar(50),
 unique (nombre)
);

create table municipio(
  id_municipio VARCHAR(4) PRIMARY KEY,
  nombre VARCHAR(100),
  id_estado int(3),
  oficina int(6),
  FOREIGN KEY (id_estado) references estado(id_estado)
);

create table municipio(
  id_municipio INT,
  nombre VARCHAR(100),
  id_estado int(3),
  oficina int(6),
  PRIMARY KEY (id_municipio, id_estado),
  FOREIGN KEY (id_estado) references estado(id_estado)
);

CREATE TABLE zona (
    id_zona INT PRIMARY KEY,
    nombre VARCHAR(30)
);

CREATE TABLE tipo_asentamiento (
    id_tipo_asentamiento INT PRIMARY KEY,
    nombre VARCHAR(40)
);

CREATE TABLE ciudad (
    id_ciudad INT PRIMARY KEY,
    nombre VARCHAR(100),
    id_municipio INT,
    FOREIGN KEY (id_municipio) REFERENCES municipio(id_municipio)
);

CREATE TABLE asentamiento (
    id_asentamiento INT PRIMARY KEY,
    nombre VARCHAR(100),
    id_tipo_asentamiento INT,
    cp_asentamiento VARCHAR(50),
    id_zona INT,
    d_codigo INT,
    id_municipio INT,
    FOREIGN KEY (id_tipo_asentamiento)
        REFERENCES tipo_asentamiento(id_tipo_asentamiento),
    FOREIGN KEY (id_zona)
        REFERENCES zona(id_zona),
    FOREIGN KEY (id_municipio)
	REFERENCES municipio(id_municipio)
);

*////////////////////////////////////////////////// Consultas (interfaces) //////////////////////////////////////////////////////*

//Estado (nombre y clave del estado)
SELECT 
    nombre AS Nombre_Estado, 
    id_estado AS Clave_Estado
FROM estado;

//Municipio (Nombre, Estado, Clave y oficina)
SELECT 
    m.nombre AS Nombre_Municipio, 
    e.nombre AS Estado, 
    m.id_municipio AS Clave_Municipio, 
    m.oficina AS Oficina
FROM municipio m, estado e
WHERE m.id_estado = e.id_estado;

//Asentamiento (Nombre, Tipo de Asentamiento, Municipio, Código de asentamiento y zona)
SELECT 
    a.nombre AS Nombre_Asentamiento, 
    ta.nombre AS Tipo_Asentamiento, 
    m.nombre AS Municipio, 
    a.d_codigo AS Codigo_Asentamiento, 
    z.nombre AS Zona
FROM asentamiento a, tipo_asentamiento ta, municipio m, zona z
WHERE a.id_tipo_asentamiento = ta.id_tipo_asentamiento
  AND a.id_municipio = m.id_municipio
  AND a.id_zona = z.id_zona;

//Zona (Nombre y Clave de la zona)
SELECT 
    nombre AS Nombre_Zona, 
    id_zona AS Clave_Zona
FROM zona;

//Consulta general (nombre asentamiento, estado, municipio ciudad, código, zona y tipo asentamiento)
SELECT 
    a.nombre AS Nombre_Asentamiento, 
    e.nombre AS Estado, 
    m.nombre AS Municipio, 
    c.nombre AS Ciudad, 
    a.d_codigo AS Codigo_Asentamiento, 
    z.nombre AS Zona, 
    ta.nombre AS Tipo_Asentamiento
FROM asentamiento a, estado e, municipio m, ciudad c, zona z, tipo_asentamiento ta
WHERE a.id_municipio = m.id_municipio
  AND m.id_estado = e.id_estado
  AND c.id_municipio = m.id_municipio
  AND a.id_zona = z.id_zona
  AND a.id_tipo_asentamiento = ta.id_tipo_asentamiento;
